\begin{frame}{Title Slide}
\phantomsection\label{title-slide}
\begin{block}{Are Harris County Drivers the Worst in Texas?}
\phantomsection\label{are-harris-county-drivers-the-worst-in-texas}
Senior project comparing selected Texas counties using crash rates,
confidence intervals, and regression.

\vspace{0.5cm}

Selected counties: Harris, Travis, Bexar, Montgomery, Hays, Medina,
Hidalgo, Hale, and Lipscomb.
\end{block}
\end{frame}

\begin{frame}{Research Question}
\phantomsection\label{research-question}
\begin{block}{Main Question}
\phantomsection\label{main-question}
Is Harris County the county with the worst drivers among the counties in
this sample?
\end{block}

\begin{block}{Working Idea}
\phantomsection\label{working-idea}
This project treats ``worst drivers'' as a question of:

\begin{itemize}
\tightlist
\item
  crash frequency relative to population
\item
  fatal crash risk
\item
  risky driving patterns such as speeding, DUI, and distraction
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Why Total Crashes Are Not Enough}
\phantomsection\label{why-total-crashes-are-not-enough}
A large county will usually have more crashes simply because it has more
people and more traffic.

That means total crashes alone can be misleading.

\begin{block}{Better comparison measures}
\phantomsection\label{better-comparison-measures}
\begin{itemize}
\tightlist
\item
  crash rate per 100,000 people
\item
  fatal crash rate per 100,000 people
\item
  proportions involving speeding, DUI, and distracted driving
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Data Source}
\phantomsection\label{data-source}
The data were gathered from the Texas Department of Transportation Motor
Vehicle Crash Statistics site:

\small

\url{https://www.txdot.gov/data-maps/crash-reports-records/motor-vehicle-crash-statistics.html}
\normalsize

The county-level values were compiled from the 2024 PDF reports for
total crashes, DUI crashes, distracted-driving crashes, speed-related
crashes, and fatal crashes.
\end{frame}

\begin{frame}{Creating New Variables}
\phantomsection\label{creating-new-variables}
To compare counties fairly, several derived variables were created.

\[
\text{Crash Rate} = \frac{\text{Total Crashes}}{\text{Population}} \times 100{,}000
\]

\[
\text{Fatal Crash Rate} = \frac{\text{Fatal Crashes}}{\text{Population}} \times 100{,}000
\]

\[
\text{Speed Percent} = \frac{\text{Speed Crashes}}{\text{Total Crashes}}
\]

\[
\text{DUI Percent} = \frac{\text{DUI Crashes}}{\text{Total Crashes}}
\]

\[
\text{Distracted Percent} = \frac{\text{Distracted Crashes}}{\text{Total Crashes}}
\]
\end{frame}

\begin{frame}{Severity Measure}
\phantomsection\label{severity-measure}
To capture how dangerous crashes are, I also used:

\[
\text{Severity Percent} = \frac{\text{Fatal Crashes} + \text{Serious Crashes}}{\text{Total Crashes}}
\]

This measure helps distinguish between counties that have many crashes
and counties that have more severe crashes.
\end{frame}

\begin{frame}{Confidence Interval Setup}
\phantomsection\label{confidence-interval-setup}
To estimate the true county crash proportion, I used:

\[
\hat{p} = \frac{\text{Total Crashes}}{\text{Population}}
\]

\[
\hat{p} \pm z \sqrt{\frac{\hat{p}(1-\hat{p})}{n}}
\]

where \(z = 1.96\) for a 95\% confidence level.
\end{frame}

\begin{frame}{Regression Model}
\phantomsection\label{regression-model}
The regression model used was:

\[
\text{Crash Rate} = \beta_0 + \beta_1(\text{SpeedPercent}) + \beta_2(\text{DUIPercent}) + \beta_3(\text{DistractedPercent}) + \varepsilon
\]

This model tests whether crash rate is associated with speeding, DUI,
and distracted-driving shares.
\end{frame}

\begin{frame}{Main Takeaways}
\phantomsection\label{main-takeaways}
\begin{itemize}
\tightlist
\item
  Total crashes are not enough to identify the ``worst drivers.''
\item
  Crash rate and fatal crash rate provide fairer county comparisons.
\item
  Risk-factor percentages help explain why counties differ.
\item
  Harris County should be judged by rate-based measures, not just size.
\end{itemize}
\end{frame}

\begin{frame}{Limitations}
\phantomsection\label{limitations}
\begin{itemize}
\tightlist
\item
  The sample includes selected counties rather than all Texas counties.
\item
  County-level analysis cannot identify individual driver behavior.
\item
  Population-based crash rates are useful, but vehicle miles traveled
  would be an even stronger exposure measure.
\item
  Small counties may produce unstable percentages because of low counts.
\end{itemize}
\end{frame}

\begin{frame}{Conclusion}
\phantomsection\label{conclusion}
This project uses rates, confidence intervals, and regression to study
whether Harris County stands out as having the worst drivers among the
selected counties.

The final answer depends on whether Harris ranks highest after adjusting
for population and whether its estimated crash rate is clearly different
from comparable counties.
\end{frame}
